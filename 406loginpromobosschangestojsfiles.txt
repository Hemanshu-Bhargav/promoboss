const express = require('express');
const router = express.Router();
const bcrypt = require('bcryptjs');
const passport = require('passport');
// const { ensureAuthenticated, forwardAuthenticated } = require('../config/auth');

//This is the model for each User or Employee entered into the database
const Users = require('../models/Users');

// Login Page
router.get('/login', (req, res) => res.render('login'));
// ForwardAuthenticated,

// Register Page
router.get('/register', (req, res) => res.render('register'));

//Register Handle
router.post('/register', (req, res) => {
    const { name, email, password, password2, empid } = req.body;
    let errors = [];
    // Check required fields, if not avail, push msg object
    if (!name || !email || !password || !password2 || empid) {
      errors.push({ msg: 'Please enter all fields' });
    }
    //Check passwords match
    if (password != password2) {
      errors.push({ msg: 'Passwords do not match' });
    }
    if (empid < 4 && empid > 6)
    {
      errors.push({msg: 'Employee ID must be 5 characters long' });
    }
  //Check password length
    if (password.length < 6) {
      errors.push({ msg: 'Password must be at least 6 characters' });
    }
    // if any of above errors are true (error len >0) then we want to rerender 
    // the register page (prompt Users to try again), otherwise (else condition) go ahead
    // and register the Users
    if (errors.length > 0) 
    { //Not only can we render register page, but also variables as well because we're using a template engine
      res.render('register', {
          //variables to pass are what needs correction from potential Users
        errors,
        //Also need to pass in data that was entered correctly
        //Without this data passed, entire register form clears itself
        name,
        email,
        password,
        password2,
        empid
      });
    } 
    else 
    {
        //Validation passed
        //Mongoose method that finds one record from MongoDb
        Users.findOne({email: email})
        .then(Users => {
            if(Users){
                //if there's a Users, rerender register form as above and send error
                errors.push({msg: 'Email is already registered'});
                res.render('register', {
                    //variables to pass are what needs correction from potential Users
                  errors,
                  //Also need to pass in data that was entered correctly
                  //Without this data passed, entire register form clears itself
                  name,
                  email,
                  password,
                  password2,
                  empid
                });
            }
            else {
                //If there's isnt an Users, create a new one
                const newUsers = new Users({
                    //pass in values for new instatiated Users
                    name, //same as name: name, just shortened because ES6 JS allows this
                    email,
                    password,
                    empid
                });
                //Hash password w/ bcrypt
                //Generate a 'salt' w/gensalt to create a hash
                bcrypt.genSalt(10, (err, salt) => 
                bcrypt.hash(newUsers.password, salt, (err, hash) => {
                    if(err) throw err;
                    newUsers.password = hash; //plain-text password is now hashed
                    //Save Users to MongoDB collection
                    newUsers.save()
                        .then(Users => {
                            //Pass express session flash error before redirect
                            req.flash('success_msg', 'Congrats! You can now access your company dashboard!');
                            res.redirect('/users/login'); //redirect to login page
                        }).catch(err => console.log(err));

                }))
            }
        });
    }

});
